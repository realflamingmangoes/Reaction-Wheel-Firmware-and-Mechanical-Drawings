25// Timer ISR for CLKIN fractional frequency generation (friend function)
26void IRAM_ATTR timer_isr_handler()
27{
28 if (g_driver_instance && g_driver_instance->clkin_half_period_us > 0.0) {
29 g_driver_instance->clkin_accumulator += 10.0; // Add 10 microsecond tick
30
31 if (g_driver_instance->clkin_accumulator >= g_driver_instance->clkin_half_period_us) {
32 g_driver_instance->clkin_accumulator -= g_driver_instance->clkin_half_period_us;
33 digitalWrite(g_driver_instance->PIN_CLKIN, !digitalRead(g_driver_instance->PIN_CLKIN));
34 }
35 }
36}

211void Driver::drv_write(uint8_t addr7, uint16_t message)
212{
213 vspi.beginTransaction(SPISettings(10000, MSBFIRST, SPI_MODE0)); // 1 MHz, Mode 0
214 scs_begin();
215 delayMicroseconds(1); // Setup time for SCS
216 vspi.transfer((0u << 7) | (addr7 & 0x7F));
217 vspi.transfer((uint8_t)(message >> 8));
218 vspi.transfer((uint8_t)(message & 0xFF));
219 delayMicroseconds(1); // Hold time for data
220 scs_end();
221 vspi.endTransaction();
222 delayMicroseconds(5); // Recovery time between transactions
223}

239uint16_t Driver::drv_read(uint8_t addr7)
240{
241 vspi.beginTransaction(SPISettings(10000, MSBFIRST, SPI_MODE0)); // Same mode as write
242 scs_begin();
243 delayMicroseconds(1); // Setup time for SCS
244 vspi.transfer((1u << 7) | (addr7 & 0x7F));
245 msb = vspi.transfer(0x00);
246 lsb = vspi.transfer(0x00);
247 delayMicroseconds(1); // Hold time for data
248 scs_end();
249 vspi.endTransaction();
250 delayMicroseconds(5); // Recovery time
251 return (uint16_t(msb) << 8) | lsb;
252}

295// ============ Interrupt-based edge detection ============ //
296void Driver::pcnt_begin_edge_isr(uint16_t threshold_edges)
297{
298 // Configure PCNT to count edges on FGOUT (PIN_FGOUT)
299 pcnt_config_t cfg = {};
300 cfg.pulse_gpio_num = PIN_FGOUT;
301 cfg.ctrl_gpio_num = PCNT_PIN_NOT_USED;
302 cfg.unit = PCNT_UNIT_USED;
303 cfg.channel = PCNT_CH_USED;
304
305 // Count BOTH edges for better resolution
306 cfg.pos_mode = PCNT_COUNT_INC; // rising
307 cfg.neg_mode = PCNT_COUNT_INC; // falling
308 cfg.lctrl_mode = PCNT_MODE_KEEP;
309 cfg.hctrl_mode = PCNT_MODE_KEEP;
310
311 cfg.counter_h_lim = 32767;
312 cfg.counter_l_lim = 0;
313
314 ESP_ERROR_CHECK(pcnt_unit_config(&cfg));
315
316 // Optional filter (APB cycles). Tune if needed.
317 ESP_ERROR_CHECK(pcnt_set_filter_value(PCNT_UNIT_USED, 80));
318 ESP_ERROR_CHECK(pcnt_filter_enable(PCNT_UNIT_USED));
319
320 // Configure threshold event
321 ESP_ERROR_CHECK(pcnt_set_event_value(PCNT_UNIT_USED, PCNT_EVT_THRES_1, (int16_t)threshold_edges));
322 ESP_ERROR_CHECK(pcnt_event_enable(PCNT_UNIT_USED, PCNT_EVT_THRES_1));
323
324 // Enable interrupts for this PCNT unit
325 ESP_ERROR_CHECK(pcnt_intr_enable(PCNT_UNIT_USED));
326
327 // Install ISR service once
328 static bool isr_service_installed = false;
329 if (!isr_service_installed) {
330 ESP_ERROR_CHECK(pcnt_isr_service_install(0));
331 isr_service_installed = true;
332 }
333
334 // Register ISR for this unit; pass `this` in case you want it later
335 ESP_ERROR_CHECK(pcnt_isr_handler_add(PCNT_UNIT_USED, Driver::pcnt_isr, this));
336
337 // Start counter
338 ESP_ERROR_CHECK(pcnt_counter_pause(PCNT_UNIT_USED));
339 ESP_ERROR_CHECK(pcnt_counter_clear(PCNT_UNIT_USED));
340 ESP_ERROR_CHECK(pcnt_counter_resume(PCNT_UNIT_USED));
341}

376void Driver::clkin_timer_begin(void)
377{
378 // Set global instance pointer for ISR
379 g_driver_instance = this;
380
381 // Initialize timer variables
382 clkin_half_period_us = 0.0; // Start with no output
383 clkin_accumulator = 0.0;
384
385 // Configure and attach Timer 1
386 // Timer 1, 80x prescaler: 80MHz / 80 = 1MHz = 1 microsecond per tick
387 clkin_timer = timerBegin(1, 80, true);
388 timerAttachInterrupt(clkin_timer, timer_isr_handler, true);
389 timerAlarmWrite(clkin_timer, 10, true); // ISR fires every 10 microseconds
390 timerAlarmEnable(clkin_timer);
391}
